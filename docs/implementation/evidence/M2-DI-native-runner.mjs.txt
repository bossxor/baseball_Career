import {createHash,randomUUID} from 'node:crypto';
import {gzipSync,gunzipSync} from 'node:zlib';
import assert from 'node:assert/strict';

const sha=b=>createHash('sha256').update(b).digest('hex'),freeze=JSON.parse(fs.readFileSync(path.join(root,'docs/implementation/evidence/M2-DI-source-freeze.json'),'utf8'));

 const event=career.events.find(e=>e.eventKey===career.phaseContext.eventKey);if(!event)throw Error('MISSING_EVENT');
 if(event.definitionId==='E43'||event.definitionId==='E41')career=engine.resolveUsWindowEvent(career,career.revision,{eventKey:event.eventKey,eventVersion:event.version,command:event.definitionId==='E43'?'FORFEIT_TRAINING':'ACK_EVENT'}).career;
 else if(['E59','E60','E61'].includes(event.definitionId))career=engine.resolveUsLevelEvent(career,career.revision,{eventKey:event.eventKey,eventVersion:event.version,command:event.definitionId==='E59'?'CONFIRM_PROMOTION':event.definitionId==='E60'?'CONFIRM_CALLUP':'CONFIRM_DEMOTION'});
 else {save('UNHANDLED_EVENT');throw Error('UNHANDLED_EVENT '+event.definitionId);}calls++;save('EVENT_RESOLVED');lastSaved=career.season.days.length;continue;}
 const r=engine.advanceUsCareer(career,career.revision);career=r.career;calls++;if(['ENGINE_LIMIT','ROSTER_DECISION_REQUIRED'].includes(r.stopReason)){save('PARTIAL_'+r.stopReason);process.exit(2);}if(career.season.days.length-lastSaved>=5||r.stopReason==='MAJOR_EVENT'){save('PARTIAL_PROGRESS');lastSaved=career.season.days.length;}
}
career=engine.settleUsSeasonGrowth(career,career.revision,true);assert.equal(Object.values(career.records.games).filter(g=>g.scope.seasonKey===career.world.seasonKey&&['ROOKIE','A','AA','AAA','MLB'].includes(g.scope.leagueKey)).length,8790);assert.equal(career.season.paidUsd,engine.usSalaryEntitlement(career,career.season.lastDay-career.season.firstDay+1));assert.equal(career.income.filter(e=>e.kind==='SIGNING_BONUS').length,1);assert.equal(career.seasonGrowth.finalized,true);save('US_REGULAR_COMPLETE');
