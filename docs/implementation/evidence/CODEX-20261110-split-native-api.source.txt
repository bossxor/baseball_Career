import fs from 'node:fs';
import assert from 'node:assert/strict';
import {createDbFromUrl,M3_CORE_SQL,gameSaves} from '@playball/db';
import {eq} from 'drizzle-orm';
import {createApp} from '../apps/api/src/app.ts';
import {readOwnedSave} from '../apps/api/src/save-state.ts';
import {careerFlowScreen} from '../apps/api/src/career-flow-screen.ts';
import {canonicalSha256} from '../packages/engine/src/canonical-seal.ts';
import {assertUsCareerRevision} from '../packages/engine/src/us-career-entry.ts';
const metadata=JSON.parse(fs.readFileSync('.local-verification/postgres-test.json','utf8').replace(/^\uFEFF/,''));
const {db,pool}=createDbFromUrl(metadata.url.replace('/postgres','/playball_profile_season_20261110'));
const id='split-api-proof-'+Date.now(),subject='profile-us',report={scope:'actual next native API command from preserved failure-boundary save; not a full season',status:'RUNNING',id};
const started=performance.now();
try {
 await pool.query(M3_CORE_SQL);
 let source=await readOwnedSave(db,subject,'profile-us');assert.ok(source);
 const original=(await pool.query("select revision::text,md5(state::text) as hash from game_saves where id='profile-us'")).rows[0];
 report.original=original;
 report.initialRecordHash=canonicalSha256(source.state.career.records.games);
 const keys=Object.keys(source.state.career.records.games);report.initialGames=keys.length;
 const action=careerFlowScreen(source.state).commands.find(c=>c.id==='adv-season');assert.ok(action);
 const revision=source.revision.toString();source=null;global.gc?.();
 await pool.query("insert into game_saves(id,owner_id,mode,revision,versions,phase,summary,state,state_format) select $1,owner_id,mode,revision,versions,phase,summary,state,state_format from game_saves where id='profile-us'",[id]);
 const app=createApp({db,auth:{allowDevBearer:true}}),headers={Authorization:'Bearer dev:'+subject,'Content-Type':'application/json'};
 const body={commandId:'boundary-native',expectedRevision:revision,type:action.type,payload:action.payload};
 const post=()=>app.request('/api/saves/'+id+'/commands',{method:'POST',headers,body:JSON.stringify(body)});
 const response=await post();report.httpStatus=response.status;
 const receipt=await response.json();assert.equal(response.status,200,JSON.stringify(receipt));
 const replay=await post();assert.equal(replay.status,200);assert.deepEqual(await replay.json(),receipt);report.replay=true;
 const stale=await app.request('/api/saves/'+id+'/commands',{method:'POST',headers,body:JSON.stringify({...body,commandId:'stale'})});assert.equal(stale.status,409);report.staleStatus=stale.status;
 let saved=await readOwnedSave(db,subject,id);assert.ok(saved);assert.equal(saved.revision.toString(),String(BigInt(revision)+1n));assert.equal(saved.stateFormat,'PARTS_V1');
 const career=saved.state.career;assertUsCareerRevision(career,career.revision);
 const prefix=Object.fromEntries(keys.map(key=>[key,career.records.games[key]]));
 assert.equal(canonicalSha256(prefix),report.initialRecordHash);report.originalGameRecordsPreserved=true;
 report.finalStateHash=canonicalSha256(saved.state);report.finalGames=Object.keys(career.records.games).length;
 report.newGames=report.finalGames-report.initialGames;assert.ok(report.newGames>0);
 const parts=(await pool.query('select count(*)::integer as parts,sum(octet_length(payload))::text as bytes,max(octet_length(payload))::integer as max_bytes from game_save_parts where save_id=$1',[id])).rows[0];report.parts=parts;
 report.originalPreserved=JSON.stringify((await pool.query("select revision::text,md5(state::text) as hash from game_saves where id='profile-us'")).rows[0])===JSON.stringify(original);assert.ok(report.originalPreserved);
 report.status='PASS';
} catch(error) {report.status='FAILED';report.error=String(error.stack??error);process.exitCode=1;}
finally {
 report.wallMs=performance.now()-started;
 fs.writeFileSync('docs/implementation/evidence/CODEX-20261110-split-native-api.json',JSON.stringify(report,null,2)+'\n');
 await db.delete(gameSaves).where(eq(gameSaves.id,id));await pool.end();console.log(JSON.stringify(report));
}
